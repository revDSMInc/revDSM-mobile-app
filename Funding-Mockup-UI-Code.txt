  document.querySelectorAll('.gbanner').forEach(b=>b.remove());

  P.forEach(p=>{

    const row = p.el, amtEl = row.querySelector('.num'), meth = row.querySelector('.meth');

    if(p.decision==='full'){
      row.classList.add('lockedrow'); row.classList.remove('needs');

      if(req > p.approvedAmt)

function stateOf(p){
  // returns index-keyed status map for the stepper
  const st = {};
  if(p.type==='check'){
    if(p.kind==='bal'){ st[0]='locked'; st[1]='locked'; }
    else {
      st[0] = p.step>=1 ? 'done' : 'now';
      st[1] = 'locked';           // rep can't deposit at the table — bank posts it
    }
  }
  if(p.type==='cc'){
    st[0] = p.linkSent ? 'done' : 'now';
    st[1] = p.step>=2 ? 'done' : (p.linkSent ? 'now' : 'next');
    st[2] = p.kind==='bal' ? 'locked' : (p.step>=3 ? 'done':'next');
  }
  if(p.type==='bank'){
    st[0] = p.linkSent ? 'done' : 'now';
    if(p.decision==='declined') st[1]='bad';
    else if(p.decision==='short') st[1] = openShort(p) ? 'warn' : 'done';
    else if(p.decision==='full') st[1]='done';
    else st[1] = p.linkSent ? 'now' : 'next';
    st[2] = p.activated ? 'done'
          : ((p.decision==='full' || (p.decision==='short' && !openShort(p))) ? 'now' : 'next');
    st[3] = p.kind==='bal' ? 'locked' : (p.step>=4?'done':'next');
  }
  return st;
}

/* how far can this payment get right now? returns null if more to do */
function doneFor(p){
  if(p.type==='check'){
    if(p.kind==='bal') return 'Collected and deposited by the install crew after the job is done.';
    if(p.step>=1) return 'Take the check to the branch to deposit it. Deposited checks off once the bank posts it.';
    return null;
  }
  if(p.type==='cc'){
    if(p.kind==='bal') return p.step>=2 ? 'Card is on file. It gets charged after install is verified.' : null;
    return p.step>=3 ? 'Charged. Funds confirmed.' : null;
  }
  if(p.type==='bank'){
    if(p.decision==='declined' || openShort(p)) return null;
    if(!p.activated) return null;
    return p.kind==='bal'
      ? 'Loan activated. Funds release after install is verified.'
      : 'Loan activated. Draw settles on the lender\'s next funding cycle.';
  }
  return null;
}

function cardHTML(p){
  const steps = FLOW[p.type] || [];
  const st = stateOf(p);
  const shortfall = (p.decision==='short' && p.requested!=null) ? p.requested - p.approvedAmt : 0;

  const stepper = steps.map((s,i)=>{
    const cls = st[i]==='next' ? '' : st[i];
    let mark = i+1;
    if(st[i]==='done') mark='✓';
    if(st[i]==='locked') mark='';
    if(st[i]==='bad') mark='✕';
    if(st[i]==='warn') mark='!';
    let sub='';
    if(i===1 && p.type==='bank' && p.approvedAmt!=null) sub='<span class="sub">$'+p.approvedAmt.toLocaleString()+'</span>';
    if(st[i]==='locked') sub='<span class="owner">'+ownerOf(p)+'</span>';
    const label = (i===1 && p.type==='bank' && p.decision==='declined') ? 'Declined' : s;
    return '<div class="step '+cls+'" onclick="tapStep('+p.id+','+i+')"><div class="dot">'+mark+'</div>'+
           '<div class="slab">'+label+sub+'</div></div>';
  }).join('');

  let extra='';
  // check: explicit rep actions

    if(p.kind==='bal'){

      extra += '';

function fundingBlockers(){
  const out = [];
  P.forEach(p=>{
    const who = '$'+p.amt.toLocaleString()+' '+p.method.split(' (')[0];
    if(p.decision==='declined'){ out.push(who+' was declined — reassign it'); return; }
    if(openShort(p)){ out.push(who+' came back short — cover the difference'); return; }
    if((p.type==='cc'||p.type==='bank') && !p.linkSent){ out.push(who+' — link not sent yet'); return; }
    if(p.type==='bank' && !p.decision){ out.push(who+' — waiting on lender decision'); return; }
    if(p.type==='bank' && p.decision && !p.activated){ out.push(who+' — loan not activated yet'); return; }
    if(p.type==='cc' && p.step<2){ out.push(who+' — card not on file yet'); return; }
    if(p.type==='cc' && p.kind==='down' && p.step<3){ out.push(who+' — card on file but not charged'); return; }
    if(p.type==='check' && p.kind==='down' && p.step<1){ out.push(who+' — check not marked collected'); return; }
  });
  const covered = P.reduce((n,p)=>n+fundedAmount(p),0);
  if(covered < PRICE) out.push('$'+(PRICE-covered).toLocaleString()+' of the contract is still unassigned');
  return out;
}

    btnEl.innerText = 'Submit Bank Decline';
    boxEl.className = 'blockers';
    boxEl.innerHTML = '<b>No funding to satisfy.</b><br>The job is closing as a bank decline.';
    return;
  }
  btnEl.innerText = 'Submit Funding';
  const b = fundingBlockers();
  const btn = document.getElementById('submitBtn');
  const box = document.getElementById('blockers');
  if(b.length){
    btn.classList.add('off'); btn.disabled = true;
    box.className = 'blockers';
    box.innerHTML = '<b>Before you can submit:</b><br>' + b.map(x=>'• '+x).join('<br>');
  } else {
    btn.classList.remove('off'); btn.disabled = false;
    box.className = 'ready';
    box.innerHTML = '✓ Every payment is as far as you can take it. Ready to submit.';
  }
}

function hasDecline(){
  return P.some(p=>p.decision==='declined') || LOG.some(l=>l.outcome==='Declined');
}

/* the Result field is locked to Sale until a lender actually declines */
function refreshResultField(){
  const ctrl = document.getElementById('resultCtrl');
  const chev = document.getElementById('resultChev');
  const hint = document.getElementById('resultHint');
  const unlocked = hasDecline();
  ctrl.classList.toggle('locked', !unlocked);
  ctrl.classList.toggle('declined', jobResult==='Bank Decline');
  chev.style.display = unlocked ? 'inline' : 'none';
  hint.style.display = unlocked ? 'none' : 'block';
  document.getElementById('resultVal').innerText = jobResult;
  document.querySelectorAll('#resultMenu .ropt').forEach(o=>{
    const on = o.dataset.v === jobResult;
    o.classList.toggle('on', on);
    o.querySelector('.rck').innerText = on ? '✓' : '';
  });
}

function setResult(v){
  jobResult = v;
  refreshResultField();
  renderFunding();
}

// wire the menu once, no inline handlers
(function(){
  const ctrl = document.getElementById('resultCtrl');
  const menu = document.getElementById('resultMenu');
  ctrl.addEventListener('click', e=>{
    e.stopPropagation();
    if(!hasDecline()) return;              // safeguard: nothing to switch to
    menu.classList.toggle('show');
    ctrl.classList.toggle('open', menu.classList.contains('show'));
  });
  menu.addEventListener('click', e=>{
    e.stopPropagation();
    const opt = e.target.closest('.ropt');
    if(!opt) return;
    menu.classList.remove('show');
    ctrl.classList.remove('open');
    setResult(opt.dataset.v);
  });
  document.addEventListener('click', ()=>{
    menu.classList.remove('show');
    ctrl.classList.remove('open');
  });
})();

  if(b.length) h+='<div class="sect-lab" style="margin-top:8px">Balance <span class="hint">Funds after install</span></div>'+b.map(cardHTML).join('');

function drawQR(){
  const c = document.getElementById('qr'), x = c.getContext('2d');
  const n=21, s=180/n; x.fillStyle='#fff'; x.fillRect(0,0,180,180); x.fillStyle='#111418';
  for(let i=0;i<n;i++) for(let j=0;j<n;j++) if(Math.random()>0.52) x.fillRect(i*s,j*s,s,s);
  [[0,0],[n-7,0],[0,n-7]].forEach(([a,b])=>{
    x.fillStyle='#fff'; x.fillRect(a*s,b*s,7*s,7*s);
    x.fillStyle='#111418'; x.fillRect(a*s,b*s,7*s,7*s);
    x.fillStyle='#fff'; x.fillRect((a+1)*s,(b+1)*s,5*s,5*s);
    x.fillStyle='#111418'; x.fillRect((a+2)*s,(b+2)*s,3*s,3*s);
  });
}

/* ---- API sync ---- */
function stampTime(){
  return new Date().toLocaleTimeString([], {hour:'numeric', minute:'2-digit'});
}
function pull(p){
  // processor confirms the homeowner saved their card
  if(p.type==='cc' && p.linkSent && p.step < 2) p.step = 2;
  if(p.type==='bank' && p.linkSent && p.queued){
    p.requested = p.amt;
    if(p.queued==='full'){ p.decision='full'; p.approvedAmt=p.amt; }
    if(p.queued==='short'){
      p.decision='short'; p.approvedAmt=Math.round(p.amt*0.6/100)*100;
      LOG.push({method:p.method, amt:p.requested, approved:p.approvedAmt, at:stampTime(), outcome:'Partial'});
    }
    if(p.queued==='declined'){
      p.decision='declined'; p.approvedAmt=0; p.activated=false;
      LOG.push({method:p.method, amt:p.requested, approved:0, at:stampTime(), outcome:'Declined'});
    }
    p.queued=null;
  }
  p.syncedAt = stampTime();
}
function syncOne(id){
  const btn = document.getElementById('cs'+id);
  if(btn) btn.classList.add('spin');
  setTimeout(()=>{ pull(P.find(x=>x.id===id)); renderFunding(); }, 750);
}

function doRefresh(){
  const btn = document.getElementById('refBtn');
  btn.classList.add('spin');
  setTimeout(()=>{
    P.forEach(pull);
    btn.classList.remove('spin');
    document.getElementById('ts').innerText = 'All synced ' + stampTime();
    renderFunding();
  }, 850);
}

      '<div class="chips">'+

      '<div class="chip'+(q==='short'?' on w':'')+'" onclick="queue('+p.id+',\'short\')">Approve partial</div>'+
      '<div class="chip'+(q==='declined'?' on b':'')+'" onclick="queue('+p.id+',\'declined\')">Decline</div>'+
      '</div>'+ (q && !p.linkSent ? '<div class="queued">Send the link first, then Refresh</div>'
                : q ? '<div class="queued">Queued — hit Sync on that card</div>' : '') + '</div>';
  }).join('');
  document.getElementById('simPanel').innerHTML =
    '<div class="sim"><h4>Scenario simulator</h4><p>Stands in for the lender API. Pick an outcome, then hit Refresh at the top to receive it.</p>'+rows+'</div>';
}
